cmake_minimum_required(VERSION 3.16)
project(uav_ommpc_ros LANGUAGES CXX)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(uav_ommpc_interfaces REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(uav_ommpc_controller CONFIG REQUIRED)

add_executable(px4_bridge src/px4_bridge.cpp)
target_compile_features(px4_bridge PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(px4_bridge PRIVATE -Wall -Wextra -Wpedantic)
endif()
ament_target_dependencies(px4_bridge rclcpp px4_msgs uav_ommpc_interfaces)
install(TARGETS px4_bridge DESTINATION lib/${PROJECT_NAME})

add_executable(ommpc_node src/ommpc_node.cpp)
target_compile_features(ommpc_node PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ommpc_node PRIVATE -Wall -Wextra -Wpedantic)
endif()
ament_target_dependencies(ommpc_node rclcpp rcl_interfaces px4_msgs uav_ommpc_interfaces)
target_link_libraries(ommpc_node uav_ommpc_controller::uav_ommpc_controller)
install(TARGETS ommpc_node DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_command_state test/test_command_state.cpp)
  target_include_directories(test_command_state PRIVATE src)
  target_compile_features(test_command_state PRIVATE cxx_std_17)
  ament_target_dependencies(test_command_state uav_ommpc_interfaces)
  ament_add_gtest(test_ommpc_inputs test/test_ommpc_inputs.cpp)
  target_include_directories(test_ommpc_inputs PRIVATE src)
  target_compile_features(test_ommpc_inputs PRIVATE cxx_std_17)
  ament_target_dependencies(test_ommpc_inputs px4_msgs uav_ommpc_interfaces)
  target_link_libraries(test_ommpc_inputs uav_ommpc_controller::uav_ommpc_controller)
endif()

ament_package()
